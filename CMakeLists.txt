cmake_minimum_required(VERSION 3.28)

# The heavy optional dependencies, each a default vcpkg manifest feature: turning one off also skips its vcpkg build.
# Read by the vcpkg toolchain during project(), so they are set before it.
option(LILA_USD "Build the OpenUSD extension: USD files, live stages, MaterialX and Hydra viewport renderers" ON)
option(LILA_VOLUMES "Build the NanoVDB extension: OpenVDB and NanoVDB volumes" ON)
if(NOT LILA_USD OR NOT LILA_VOLUMES)
    set(VCPKG_MANIFEST_NO_DEFAULT_FEATURES ON)
    set(VCPKG_MANIFEST_FEATURES "")
    if(LILA_USD)
        list(APPEND VCPKG_MANIFEST_FEATURES usd)
    endif()
    if(LILA_VOLUMES)
        list(APPEND VCPKG_MANIFEST_FEATURES volumes)
    endif()
endif()
# Pins the MSVC toolset for dependencies (vcpkg otherwise picks the newest installed Visual Studio).
set(VCPKG_OVERLAY_TRIPLETS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/triplets")

project(Lila VERSION 0.1.0 LANGUAGES C CXX)

if(MSVC AND CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/triplets/x64-windows.cmake" _lila_toolset REGEX "VCPKG_PLATFORM_TOOLSET_VERSION")
    string(REGEX MATCH "14\\.([0-9]+)" _lila_toolset "${_lila_toolset}")
    math(EXPR _lila_msvc "1900 + ${CMAKE_MATCH_1}")
    if(NOT MSVC_VERSION EQUAL _lila_msvc)
        message(FATAL_ERROR "Lila builds with MSVC ${_lila_toolset} (cmake/triplets/x64-windows.cmake), this is ${MSVC_VERSION}. "
            "Build with build.bat, which sets it up and reconfigures this build folder (${CMAKE_BINARY_DIR}). "
            "If ${_lila_toolset} is missing, add it in the Visual Studio Installer.")
    endif()
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# No C++20 modules: scanning costs a preprocessor pass per source and keeps sources out of unity builds.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

option(LILA_BUILD_TESTS "Build unit tests" ON)

if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8 /Zc:__cplusplus)
    if(NOT CMAKE_GENERATOR MATCHES "Ninja")
        add_compile_options(/MP)
    endif()
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    # Lila's DLLs export classes holding STL members and deriving from Qt and std types. That is the SDK's ABI (same
    # compiler and runtime on both sides), so the warnings about it say nothing.
    add_compile_options(/wd4251 /wd4275)
endif()

list(APPEND CMAKE_MODULE_PATH ${CMAKE_SOURCE_DIR}/cmake)

find_package(Python 3.14 EXACT REQUIRED COMPONENTS Interpreter Development.Module Development.Embed)
# Qt is vcpkg's, also in build folders whose cache still points at a Qt installed elsewhere.
if(VCPKG_INSTALLED_DIR)
    get_cmake_property(_cache_vars CACHE_VARIABLES)
    foreach(_var IN LISTS _cache_vars)
        if(_var MATCHES "^Qt6.+_DIR$" AND NOT "${${_var}}" MATCHES "^${VCPKG_INSTALLED_DIR}")
            unset(${_var} CACHE)
        endif()
    endforeach()
    set(Qt6_DIR "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/Qt6" CACHE PATH "" FORCE)
endif()
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Svg Network)
include(LilaDeployQt)
include(LilaPySide)
# Q_SIGNALS, Q_SLOTS and Q_EMIT only: the precompiled header puts Qt in every source, where signals, slots and emit are
# ordinary names.
add_compile_definitions(QT_NO_KEYWORDS)
find_package(TBB CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(volk CONFIG REQUIRED)
find_package(VulkanHeaders CONFIG REQUIRED)
find_package(VulkanMemoryAllocator CONFIG REQUIRED)
find_package(httplib CONFIG REQUIRED)
# Help > Record Performance Trace. The client (vcpkg-ports/tracy) listens on 127.0.0.1 only and records nothing until a capture connects.
option(LILA_NO_TRACY "Build without the Tracy profiler" OFF)
unset(LILA_TRACY CACHE)
set(LILA_TRACY ON)
if(LILA_NO_TRACY)
    set(LILA_TRACY OFF)
endif()
if(LILA_TRACY)
    find_package(Tracy CONFIG REQUIRED)
endif()
find_package(doctest CONFIG REQUIRED)
find_package(Git QUIET)

include(LilaDeps)

include(LilaExtension)
include(LilaBuildSpeed)
include(LilaVersion)
